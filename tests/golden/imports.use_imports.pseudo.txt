// use_imports
extern uint64_t g_count;
extern FILE* stdout;

void use_imports(const char* a, const char* b) {
  if (a) {
    uint64_t r_strlen = strlen(a);
    g_count += r_strlen;
    puts(a);
  }
  if (b) {
    fputs(b, stdout);
  }
  char* r_getenv = getenv("HOME");
  if (!r_getenv) {
    return;
  }
  g_count++;
  return;
}
