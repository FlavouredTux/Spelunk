// <entry>
typedef union PpcState {
  struct { uint32_t word_0; };
  struct { uint8_t _pad_byte_4[4]; uint8_t byte_4; };
  struct { uint8_t _pad_half_6[6]; uint16_t half_6; };
} PpcState;

uint32_t scalar_entry(PpcState* state) {
  if (!state) {
    return 0;
  }
  uint32_t value = (uint32_t)state->byte_4;
  return state->word_0 * 3 + value * 4 + (uint32_t)state->half_6;
}
